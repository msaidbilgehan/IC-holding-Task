//// -- Tables and References

// Creating tables
Table Person {
  id int [pk, increment] // auto-increment
  first_name varchar [not null]
  last_name varchar [not null]
}

Table Task {
  id int [pk, increment] // auto-increment
  title varchar [not null]
  description text
}

Table TaskAssignment {
  id int [pk, increment] // auto-increment
  task_id int [not null]
  person_id int [not null]

  // Indexing
  Indexes {
    (task_id, person_id) [name:"index_task_person_assignment"]
    id [unique]
  }
}

Table TaskCompletionLog {
  id int [pk, increment] // auto-increment
  task_assignment_id int [not null]
  time_start timestamp [not null]
  time_completed timestamp

  // Indexing
  Indexes {
    (task_assignment_id) [name:"index_task_person_completion_log"]
    id [unique]
  }
}


// Creating references
Ref: Person.id < TaskAssignment.person_id
Ref: Task.id < TaskAssignment.task_id

Ref: TaskAssignment.id < TaskCompletionLog.task_assignment_id
